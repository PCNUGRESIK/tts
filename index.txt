  <script>
    'use strict';

    /* ============================================================
       1. SECURITY & IMAGE OPTIMIZATION UTILITIES
       ============================================================ */
    const Security = {
      /**
       * Escape HTML special characters to prevent XSS injection.
       */
      escapeHtml(val) {
        if (val === null || val === undefined) return '';
        return String(val)
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;')
          .replace(/'/g, '&#39;');
      },

      /**
       * Allow only safe image data URLs or http(s) URLs.
       */
      sanitizeImageSrc(src) {
        const s = String(src || '').trim();
        if (!s) return '';
        if (/^data:image\/(png|jpeg|jpg|webp|gif|svg\+xml);base64,[A-Za-z0-9+/=]+$/i.test(s)) {
          return s;
        }
        if (/^https?:\/\/[^\s"'<>]+$/i.test(s)) {
          return s;
        }
        return '';
      },

      /**
       * Generate collision-resistant IDs for records, menus, fields, and users.
       */
      generateId(prefix = 'id') {
        const rand = Math.random().toString(36).slice(2, 8);
        return `${prefix}_${Date.now().toString(36)}_${rand}`;
      },

      /**
       * Validate uploaded Kop image MIME type & size (max 5 MB raw before compression).
       */
      validateImageFile(file) {
        const allowed = ['image/png', 'image/jpeg', 'image/webp'];
        if (!file) return { ok: false, error: 'File tidak ditemukan.' };
        if (!allowed.includes(file.type)) {
          return { ok: false, error: 'Format gambar tidak didukung. Gunakan PNG, JPG, atau WebP.' };
        }
        if (file.size > 5 * 1024 * 1024) {
          return { ok: false, error: 'Ukuran gambar terlalu besar (maksimal 5 MB).' };
        }
        return { ok: true };
      },

      /**
       * Compress Kop Header image via Canvas so LocalStorage & Drive uploads stay fast and sharp.
       */
      compressImageToDataUrl(file, maxWidth = 1400, quality = 0.88) {
        return new Promise((resolve, reject) => {
          const reader = new FileReader();
          reader.onerror = () => reject(new Error('Gagal membaca file gambar.'));
          reader.onload = (ev) => {
            const img = new Image();
            img.onerror = () => reject(new Error('File bukan gambar yang valid.'));
            img.onload = () => {
              const ratio = img.width > maxWidth ? maxWidth / img.width : 1;
              const targetW = Math.max(1, Math.round(img.width * ratio));
              const targetH = Math.max(1, Math.round(img.height * ratio));
              const canvas = document.createElement('canvas');
              canvas.width = targetW;
              canvas.height = targetH;
              const ctx = canvas.getContext('2d');
              ctx.clearRect(0, 0, targetW, targetH);
              ctx.drawImage(img, 0, 0, targetW, targetH);
              const outMime = file.type === 'image/png' ? 'image/png' : 'image/jpeg';
              const dataUrl = canvas.toDataURL(outMime, quality);
              resolve({ dataUrl, mimeType: outMime });
            };
            img.src = ev.target.result;
          };
          reader.readAsDataURL(file);
        });
      }
    };

    /* ============================================================
       2. RBAC PERMISSION CATALOG & AUTH MODULE (SUPERADMIN / OPERATOR)
       ============================================================ */
    const ROLES = {
      SUPERADMIN: 'superadmin',
      OPERATOR: 'operator'
    };

    const PERMISSION_DEFS = [
      { key: 'record.view', label: 'Lihat Dashboard & Arsip', group: 'Tanda Terima', operatorAllowed: true, defaultOperator: true },
      { key: 'record.create', label: 'Buat Tanda Terima Baru', group: 'Tanda Terima', operatorAllowed: true, defaultOperator: true },
      { key: 'record.edit', label: 'Ubah Tanda Terima', group: 'Tanda Terima', operatorAllowed: true, defaultOperator: true },
      { key: 'record.delete', label: 'Hapus Arsip Tanda Terima', group: 'Tanda Terima', operatorAllowed: true, defaultOperator: false },
      { key: 'record.print', label: 'Cetak Format A4 / 2×A5', group: 'Tanda Terima', operatorAllowed: true, defaultOperator: true },
      { key: 'kop.manage', label: 'Ubah Kop Gambar & Tata Letak A4/A5', group: 'Pengaturan', operatorAllowed: true, defaultOperator: false },
      { key: 'menu.manage', label: 'Atur Menu, Field, Tabel & TTD Dinamis', group: 'Pengaturan', operatorAllowed: true, defaultOperator: false },
      { key: 'system.sync', label: 'Integrasi GAS, Backup & Restore JSON', group: 'Sistem', operatorAllowed: true, defaultOperator: false },
      { key: 'user.manage', label: 'Kelola User & Hak Akses (Khusus Superadmin)', group: 'Keamanan', operatorAllowed: false, defaultOperator: false }
    ];

    const ALL_PERMISSION_KEYS = PERMISSION_DEFS.map((p) => p.key);
    const DEFAULT_OPERATOR_PERMS = PERMISSION_DEFS.filter((p) => p.defaultOperator).map((p) => p.key);

    const MENU_PERMISSION_MAP = {
      dashboard: 'record.view',
      create: 'record.create',
      archive: 'record.view',
      kop: 'kop.manage',
      menus: 'menu.manage',
      integration: 'system.sync',
      users: 'user.manage'
    };

    const Auth = {
      STORAGE_KEYS: {
        LOCAL_USERS: 'tt_a4_v2_local_users',
        SESSION: 'tt_a4_v2_session'
      },
      _session: null,

      async hashLocalPassword(password, salt) {
        const input = `${salt}:${password}:tt_local_v2`;
        if (window.crypto && window.crypto.subtle) {
          try {
            const data = new TextEncoder().encode(input);
            const buf = await window.crypto.subtle.digest('SHA-256', data);
            return Array.from(new Uint8Array(buf)).map((b) => b.toString(16).padStart(2, '0')).join('');
          } catch (_) {}
        }
        let h = 2166136261;
        for (let i = 0; i < input.length; i++) {
          h ^= input.charCodeAt(i);
          h = Math.imul(h, 16777619);
        }
        return 'fnv_' + (h >>> 0).toString(16);
      },

      sanitizePermissions(role, perms) {
        if (role === ROLES.SUPERADMIN) return [...ALL_PERMISSION_KEYS];
        const list = Array.isArray(perms) ? perms : DEFAULT_OPERATOR_PERMS;
        return Array.from(new Set(list.filter((p) => ALL_PERMISSION_KEYS.includes(p) && p !== 'user.manage')));
      },

      getSession() {
        if (this._session) return this._session;
        try {
          const raw = sessionStorage.getItem(this.STORAGE_KEYS.SESSION);
          if (!raw) return null;
          const parsed = JSON.parse(raw);
          if (!parsed || !parsed.user) return null;
          if (parsed.expiresAt && new Date(parsed.expiresAt).getTime() <= Date.now()) {
            this.clearSession();
            return null;
          }
          this._session = parsed;
          return parsed;
        } catch (_) {
          return null;
        }
      },

      saveSession(sessionObj) {
        this._session = sessionObj;
        try {
          sessionStorage.setItem(this.STORAGE_KEYS.SESSION, JSON.stringify(sessionObj));
        } catch (_) {}
      },

      clearSession() {
        this._session = null;
        try {
          sessionStorage.removeItem(this.STORAGE_KEYS.SESSION);
        } catch (_) {}
      },

      isAuthenticated() {
        return Boolean(this.getSession() && this.getSession().user);
      },

      getCurrentUser() {
        const s = this.getSession();
        return s ? s.user : null;
      },

      getSessionToken() {
        const s = this.getSession();
        return (s && s.sessionToken) || '';
      },

      isSuperadmin() {
        const u = this.getCurrentUser();
        return Boolean(u && u.role === ROLES.SUPERADMIN);
      },

      can(perm) {
        const u = this.getCurrentUser();
        if (!u) return false;
        if (u.role === ROLES.SUPERADMIN) return true;
        return Array.isArray(u.permissions) && u.permissions.includes(perm);
      },

      canAccessMenu(menu) {
        if (!menu) return false;
        if (this.isSuperadmin()) return true;
        if (menu.type === 'custom_filter') return this.can('record.view');
        if (menu.type === 'custom_page') return true;
        const reqPerm = menu.requiredPerm || MENU_PERMISSION_MAP[menu.id];
        if (!reqPerm) return true;
        if (menu.id === 'create') return this.can('record.create') || this.can('record.edit');
        return this.can(reqPerm);
      },

      getLocalUsers() {
        try {
          const raw = localStorage.getItem(this.STORAGE_KEYS.LOCAL_USERS);
          if (!raw) return [];
          const parsed = JSON.parse(raw);
          return Array.isArray(parsed) ? parsed : [];
        } catch (_) {
          return [];
        }
      },

      _commitLocalUsers(users) {
        localStorage.setItem(this.STORAGE_KEYS.LOCAL_USERS, JSON.stringify(users));
      },

      hasLocalUsers() {
        return this.getLocalUsers().length > 0;
      },

      async setupFirstLocalSuperadmin({ username, fullName, password }) {
        const cleanUser = String(username || '').trim().toLowerCase();
        if (!/^[a-z0-9._-]{3,32}$/.test(cleanUser)) {
          return { ok: false, error: 'Username harus 3–32 karakter (huruf kecil, angka, titik, underscore).' };
        }
        if (!String(fullName || '').trim()) {
          return { ok: false, error: 'Nama lengkap wajib diisi.' };
        }
        if (String(password || '').length < 6) {
          return { ok: false, error: 'Kata sandi minimal 6 karakter.' };
        }
        const salt = Security.generateId('salt');
        const hash = await this.hashLocalPassword(password, salt);
        const now = new Date().toISOString();
        const user = {
          id: Security.generateId('usr'),
          username: cleanUser,
          fullName: String(fullName).trim(),
          role: ROLES.SUPERADMIN,
          permissions: [...ALL_PERMISSION_KEYS],
          passwordSalt: salt,
          passwordHash: hash,
          active: true,
          createdAt: now,
          updatedAt: now
        };
        this._commitLocalUsers([user]);
        const safeUser = {
          id: user.id, username: user.username, fullName: user.fullName,
          role: user.role, permissions: user.permissions, active: true
        };
        this.saveSession({
          mode: 'local',
          sessionToken: '',
          expiresAt: new Date(Date.now() + 12 * 3600 * 1000).toISOString(),
          user: safeUser
        });
        return { ok: true, user: safeUser };
      },

      async loginLocal(username, password) {
        const cleanUser = String(username || '').trim().toLowerCase();
        const users = this.getLocalUsers();
        const found = users.find((u) => u.username === cleanUser);
        if (!found) return { ok: false, error: 'Username atau kata sandi salah.' };
        if (!found.active) return { ok: false, error: 'Akun Anda sedang dinonaktifkan oleh Superadmin.' };

        const computed = await this.hashLocalPassword(password, found.passwordSalt);
        if (computed !== found.passwordHash) {
          return { ok: false, error: 'Username atau kata sandi salah.' };
        }
        const safeUser = {
          id: found.id,
          username: found.username,
          fullName: found.fullName,
          role: found.role,
          permissions: this.sanitizePermissions(found.role, found.permissions),
          active: true
        };
        this.saveSession({
          mode: 'local',
          sessionToken: '',
          expiresAt: new Date(Date.now() + 12 * 3600 * 1000).toISOString(),
          user: safeUser
        });
        return { ok: true, user: safeUser };
      },

      async upsertLocalUser(payload) {
        if (!this.can('user.manage')) return { ok: false, error: 'Akses ditolak: khusus Superadmin.' };
        const users = [...this.getLocalUsers()];
        const username = String(payload.username || '').trim().toLowerCase();
        const fullName = String(payload.fullName || '').trim();
        const role = payload.role === ROLES.SUPERADMIN ? ROLES.SUPERADMIN : ROLES.OPERATOR;
        const active = payload.active !== undefined ? Boolean(payload.active) : true;
        const permissions = this.sanitizePermissions(role, payload.permissions);
        const now = new Date().toISOString();

        if (!/^[a-z0-9._-]{3,32}$/.test(username)) {
          return { ok: false, error: 'Username harus 3–32 karakter (huruf kecil, angka, titik, underscore).' };
        }
        if (!fullName) return { ok: false, error: 'Nama lengkap wajib diisi.' };

        const existingIdx = payload.id ? users.findIndex((u) => u.id === payload.id) : -1;
        const duplicate = users.find((u, i) => u.username === username && i !== existingIdx);
        if (duplicate) return { ok: false, error: `Username "${username}" sudah digunakan.` };

        const activeSuperCount = users.filter((u) => u.role === ROLES.SUPERADMIN && u.active).length;

        if (existingIdx >= 0) {
          const prev = users[existingIdx];
          const wasActiveSuper = prev.role === ROLES.SUPERADMIN && prev.active;
          const willBeActiveSuper = role === ROLES.SUPERADMIN && active;
          if (wasActiveSuper && !willBeActiveSuper && activeSuperCount <= 1) {
            return { ok: false, error: 'Tidak dapat menonaktifkan atau mengubah role Superadmin aktif terakhir.' };
          }
          let salt = prev.passwordSalt;
          let hash = prev.passwordHash;
          if (payload.password) {
            if (payload.password.length < 6) return { ok: false, error: 'Kata sandi baru minimal 6 karakter.' };
            salt = Security.generateId('salt');
            hash = await this.hashLocalPassword(payload.password, salt);
          }
          users[existingIdx] = {
            ...prev,
            username,
            fullName,
            role,
            permissions,
            passwordSalt: salt,
            passwordHash: hash,
            active,
            updatedAt: now
          };
          this._commitLocalUsers(users);
          const cur = this.getCurrentUser();
          if (cur && cur.id === prev.id) {
            const s = this.getSession();
            this.saveSession({
              ...s,
              user: { id: prev.id, username, fullName, role, permissions, active }
            });
          }
          return { ok: true };
        }

        if (!payload.password || payload.password.length < 6) {
          return { ok: false, error: 'Kata sandi akun baru minimal 6 karakter.' };
        }
        const salt = Security.generateId('salt');
        const hash = await this.hashLocalPassword(payload.password, salt);
        users.push({
          id: Security.generateId('usr'),
          username,
          fullName,
          role,
          permissions,
          passwordSalt: salt,
          passwordHash: hash,
          active,
          createdAt: now,
          updatedAt: now
        });
        this._commitLocalUsers(users);
        return { ok: true };
      },

      deleteLocalUser(userId) {
        if (!this.can('user.manage')) return { ok: false, error: 'Akses ditolak.' };
        const cur = this.getCurrentUser();
        if (cur && cur.id === userId) return { ok: false, error: 'Tidak dapat menghapus akun sendiri.' };
        const users = this.getLocalUsers();
        const target = users.find((u) => u.id === userId);
        if (!target) return { ok: false, error: 'Akun tidak ditemukan.' };
        const activeSupers = users.filter((u) => u.role === ROLES.SUPERADMIN && u.active).length;
        if (target.role === ROLES.SUPERADMIN && target.active && activeSupers <= 1) {
          return { ok: false, error: 'Tidak dapat menghapus Superadmin aktif terakhir.' };
        }
        this._commitLocalUsers(users.filter((u) => u.id !== userId));
        return { ok: true };
      },

      async changeLocalOwnPassword(currentPassword, newPassword) {
        const cur = this.getCurrentUser();
        if (!cur) return { ok: false, error: 'Sesi tidak valid.' };
        if (String(newPassword || '').length < 6) return { ok: false, error: 'Kata sandi baru minimal 6 karakter.' };
        const users = [...this.getLocalUsers()];
        const idx = users.findIndex((u) => u.id === cur.id);
        if (idx < 0) return { ok: false, error: 'Akun tidak ditemukan.' };
        const check = await this.hashLocalPassword(currentPassword, users[idx].passwordSalt);
        if (check !== users[idx].passwordHash) return { ok: false, error: 'Kata sandi lama tidak sesuai.' };
        const newSalt = Security.generateId('salt');
        users[idx].passwordSalt = newSalt;
        users[idx].passwordHash = await this.hashLocalPassword(newPassword, newSalt);
        users[idx].updatedAt = new Date().toISOString();
        this._commitLocalUsers(users);
        return { ok: true };
      }
    };

      async _syncToGas(action, payload, isPublic = false) {
        const cfg = this.getConfig();
        if (!cfg.gas.endpointUrl) {
          return { ok: false, code: 'NO_ENDPOINT', error: 'Endpoint Web App GAS belum diatur.' };
        }
        const reqKey = `${action}:${JSON.stringify(payload || {}).slice(0, 80)}`;
        if (this._inFlight.has(reqKey)) return this._inFlight.get(reqKey);

        const controller = new AbortController();
        const timer = setTimeout(() => controller.abort(), 18000);

        const bodyPayload = {
          action,
          sessionToken: isPublic ? '' : Auth.getSessionToken(),
          payload: payload || {}
        };

        const promise = fetch(cfg.gas.endpointUrl, {
          method: 'POST',
          headers: { 'Content-Type': 'text/plain;charset=utf-8' },
          body: JSON.stringify(bodyPayload),
          signal: controller.signal
        })
          .then(async (resp) => {
            clearTimeout(timer);
            if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
            const json = await resp.json();
            if (!isPublic && json && (json.code === 'UNAUTHORIZED' || json.code === 'SESSION_EXPIRED')) {
              Auth.clearSession();
              if (typeof App !== 'undefined' && App.showAuthScreen) {
                App.showAuthScreen('Sesi berakhir. Silakan login kembali.');
              }
            }
            return json;
          })
          .catch((err) => {
            clearTimeout(timer);
            if (err && err.name === 'AbortError') {
              return { ok: false, code: 'TIMEOUT', error: 'Server GAS tidak merespons (timeout 18 detik).' };
            }
            return { ok: false, code: 'NETWORK_ERROR', error: (err && err.message) || 'Gagal menghubungi server GAS.' };
          })
          .finally(() => {
            this._inFlight.delete(reqKey);
          });

        this._inFlight.set(reqKey, promise);
        return promise;
      },

      printRecord(record, overridePrintMode = null) {
        if (!Auth.can('record.print')) {

          return;
        }
        const cfg = DataService.getConfig();
        const portal = document.getElementById('print-portal');
        portal.innerHTML = this.renderSheetHtml(record, cfg, overridePrintMode);
        const imgs = portal.querySelectorAll('img');
        const waitImages = Array.from(imgs).map((img) => {
          if (img.complete) return Promise.resolve();
          return new Promise((res) => {
            img.onload = res;
            img.onerror = res;
          });
        });
        Promise.all(waitImages).then(() => {
          setTimeout(() => window.print(), 80);
        });

        const enabled = Boolean(document.getElementById('auth-gas-enabled')?.checked);

        cfg.gas.enabled = enabled;

      async handleFirstSuperadminSubmit(ev) {
        ev.preventDefault();
        const fullName = document.getElementById('setup-fullname').value.trim();
        const username = document.getElementById('setup-username').value.trim();
        const pwd = document.getElementById('setup-password').value;
        const pwd2 = document.getElementById('setup-password2').value;
        if (pwd !== pwd2) {
          UI.toast('Konfirmasi kata sandi tidak cocok.', 'warning');
          return;
        }
        const res = await Auth.setupFirstLocalSuperadmin({ username, fullName, password: pwd });
        if (!res.ok) {
          UI.toast(res.error, 'error');
          return;
        }
        UI.toast(`Selamat datang, ${res.user.fullName} (Superadmin)!`, 'success');
        this.bootAuthenticatedApp();
      },

      async handleLoginSubmit(ev) {
        ev.preventDefault();
        const username = document.getElementById('login-username').value.trim();
        const password = document.getElementById('login-password').value;
        const btn = document.getElementById('auth-submit-btn');
        if (btn) {
          btn.disabled = true;
          btn.textContent = 'Memverifikasi...';
        }
        try {
          const cfg = DataService.getConfig();
          const isGasMode = Boolean(cfg.gas.enabled && cfg.gas.endpointUrl);
          const res = isGasMode
            ? await DataService.gasLogin(username, password)
            : await Auth.loginLocal(username, password);

          if (!res.ok) {
            UI.toast(res.error || 'Gagal login.', 'error');
            return;
          }

          if (isGasMode) {
            UI.toast('Login berhasil. Memuat data dari Spreadsheet...', 'info');
            await DataService.bootstrapFromGas();
          }
          UI.toast(`Selamat datang, ${res.user.fullName} (${res.user.role.toUpperCase()}).`, 'success');
          this.bootAuthenticatedApp();
        } finally {
          if (btn) {
            btn.disabled = false;
            btn.textContent = 'Masuk';
          }
        }
      },

      navigateTo(menuId, options = {}) {
        if (!Auth.isAuthenticated()) {
          this.showAuthScreen();
          return;
        }
        const cfg = DataService.getConfig();
        let targetMenu = cfg.menus.find((m) => m.id === menuId);
        if (!targetMenu) {
          targetMenu = cfg.menus.find((m) => m.visible && Auth.canAccessMenu(m)) || cfg.menus[0];
          menuId = targetMenu.id;
        }

        this.state.activeMenuId = menuId;
        this.toggleSidebar(false);
        this.renderSidebar();
        this.renderUserBadge();

        document.getElementById('topbar-title').textContent = targetMenu.label;
        document.getElementById('topbar-desc').textContent = targetMenu.description || '';

        const main = document.getElementById('main-view');

        if (!Auth.canAccessMenu(targetMenu)) {
          main.innerHTML = `
            <div class="bg-white rounded-2xl p-10 border border-rose-200 text-center max-w-lg mx-auto mt-8 shadow-xs">
              <div class="w-14 h-14 rounded-2xl bg-rose-50 text-rose-600 mx-auto flex items-center justify-center mb-3">

              <p class="text-sm text-slate-600 mt-1">Akun Anda tidak memiliki hak akses untuk membuka menu <strong>${Security.escapeHtml(targetMenu.label)}</strong>. Hubungi Superadmin jika Anda memerlukan izin ini.</p>
            </div>
          `;
          lucide.createIcons({ nodes: [main] });
          return;
        }

        if (menuId === 'dashboard') {
          this.renderDashboardView(main);
        } else if (menuId === 'create') {
          if (options.resetDraft || !this.state.editorDraft) {
            this.initNewDraft();
          }
          this.renderEditorView(main);
        } else if (menuId === 'archive') {
          this.renderArchiveView(main, options.presetStatus || null);
        } else if (menuId === 'kop') {
          this.renderKopA4View(main);
        } else if (menuId === 'menus') {
          this.renderDynamicSettingsView(main);
        } else if (menuId === 'users') {
          this.renderUsersView(main);
        } else if (menuId === 'integration') {
          this.renderIntegrationView(main);
        } else if (targetMenu.type === 'custom_filter') {
          this.renderArchiveView(main, targetMenu.filterStatus || 'ALL', targetMenu);
        } else if (targetMenu.type === 'custom_page') {
          this.renderCustomPageView(main, targetMenu);
        } else {
          this.renderDashboardView(main);
        }

        lucide.createIcons({ nodes: [main] });
      },

      initNewDraft() {
        const cfg = DataService.getConfig();
        const meta = {};
        cfg.metaFields.forEach((f) => {
          meta[f.id] = f.type === 'date' ? Format.todayISO() : (f.defaultValue || '');
        });
        const firstRow = {};
        cfg.tableColumns.forEach((c) => {
          firstRow[c.id] = c.id === 'jumlah' ? '1' : c.id === 'satuan' ? 'Berkas' : '';
        });
        this.state.editingRecordId = null;
        this.state.editorDraft = {
          id: null,
          nomor: DataService.generateNextNumber(),
          status: 'Diterima',
          meta,
          items: [firstRow],
          notes: '',
          signatures: {}
        };
      },

      editRecord(id) {
        if (!Auth.can('record.edit')) {
          UI.toast('Akses ditolak: Anda tidak memiliki izin mengubah tanda terima.', 'error');
          return;
        }
        const rec = DataService.getRecords().find((r) => r.id === id);
        if (!rec) {
          UI.toast('Data tanda terima tidak ditemukan.', 'error');
          return;
        }
        this.state.editingRecordId = rec.id;
        this.state.editorDraft = JSON.parse(JSON.stringify(rec));
        this.navigateTo('create');
      },

      setQuickPrintMode(mode) {
        const cfg = DataService.getConfig();
        cfg.a4.printMode = mode;
        DataService.saveConfig(cfg);
        this.updateStorageBadge();
        this.renderEditorView(document.getElementById('main-view'));
        lucide.createIcons({ nodes: [document.getElementById('main-view')] });
      },

      setMobileEditorTab(tab) {
        this.state.mobileEditorTab = tab;
        this.renderEditorView(document.getElementById('main-view'));
        lucide.createIcons({ nodes: [document.getElementById('main-view')] });
      },

      refreshEditorA4Preview() {
        const mount = document.getElementById('editor-a4-mount');
        if (!mount || !this.state.editorDraft) return;
        const cfg = DataService.getConfig();
        mount.innerHTML = A4Renderer.renderSheetHtml(this.state.editorDraft, cfg);
        const sheet = mount.querySelector('.a4-sheet');
        if (sheet) sheet.classList.add('a4-sheet-screen');
        requestAnimationFrame(() => A4Renderer.fitPreviewToContainer('editor-a4-wrapper'));
      },

      updateDraftRoot(key, value) {
        if (!this.state.editorDraft) return;
        this.state.editorDraft[key] = value;
        this.refreshEditorA4Preview();
      },

      updateDraftMeta(fieldId, value) {
        if (!this.state.editorDraft) return;
        this.state.editorDraft.meta[fieldId] = value;
        this.refreshEditorA4Preview();
      },

      updateDraftItem(idx, colId, value) {
        if (!this.state.editorDraft || !this.state.editorDraft.items[idx]) return;
        this.state.editorDraft.items[idx][colId] = value;
        this.refreshEditorA4Preview();
      },

      updateDraftSignature(sigId, prop, value) {
        if (!this.state.editorDraft) return;
        if (!this.state.editorDraft.signatures) this.state.editorDraft.signatures = {};
        if (!this.state.editorDraft.signatures[sigId]) this.state.editorDraft.signatures[sigId] = {};
        this.state.editorDraft.signatures[sigId][prop] = value;
        this.refreshEditorA4Preview();
      },

      addDraftItem() {
        const cfg = DataService.getConfig();
        const row = {};
        cfg.tableColumns.forEach((c) => {
          row[c.id] = c.id === 'jumlah' ? '1' : c.id === 'satuan' ? 'Berkas' : '';
        });
        this.state.editorDraft.items.push(row);
        this.renderEditorView(document.getElementById('main-view'));
        lucide.createIcons({ nodes: [document.getElementById('main-view')] });
      },

      removeDraftItem(idx) {
        if (this.state.editorDraft.items.length <= 1) return;
        this.state.editorDraft.items.splice(idx, 1);
        this.renderEditorView(document.getElementById('main-view'));
        lucide.createIcons({ nodes: [document.getElementById('main-view')] });
      },

      regenerateDraftNumber() {
        this.state.editorDraft.nomor = DataService.generateNextNumber();
        this.renderEditorView(document.getElementById('main-view'));
        lucide.createIcons({ nodes: [document.getElementById('main-view')] });
        UI.toast('Nomor tanda terima diperbarui.', 'info');
      },

      resetEditorDraft() {
        UI.confirm({
          title: 'Reset Formulir?',
          message: 'Semua isian yang belum disimpan pada formulir ini akan dikembalikan ke kosong.',
          confirmText: 'Ya, Reset',
          onConfirm: () => {
            this.initNewDraft();
            this.renderEditorView(document.getElementById('main-view'));
            lucide.createIcons({ nodes: [document.getElementById('main-view')] });
            UI.toast('Formulir berhasil direset.', 'info');
          }
        });
      },

      async saveCurrentDraft(andPrint = false) {
        const isEditing = Boolean(this.state.editingRecordId);
        if (isEditing && !Auth.can('record.edit')) {
          UI.toast('Akses ditolak: Anda tidak memiliki izin mengubah data.', 'error');
          return;
        }
        if (!isEditing && !Auth.can('record.create')) {
          UI.toast('Akses ditolak: Anda tidak memiliki izin membuat tanda terima baru.', 'error');
          return;
        }
        if (andPrint && !Auth.can('record.print')) {
          UI.toast('Akses ditolak: Anda tidak memiliki izin cetak.', 'warning');
          andPrint = false;
        }
        const draft = this.state.editorDraft;
        const cfg = DataService.getConfig();

        if (!draft.nomor || !draft.nomor.trim()) {
          UI.toast('Nomor tanda terima wajib diisi.', 'warning');
          return;
        }
        for (const field of cfg.metaFields) {
          if (field.visible && field.required && !(draft.meta && String(draft.meta[field.id] || '').trim())) {
            UI.toast(`Kolom "${field.label}" wajib diisi.`, 'warning');
            return;
          }
        }

        const res = await DataService.saveRecord(draft);
        if (!res.ok) {
          UI.toast(res.error || 'Gagal menyimpan data.', 'error');
          return;
        }

        this.state.editingRecordId = res.data.id;
        this.state.editorDraft = JSON.parse(JSON.stringify(res.data));
        this.renderSidebar();
        UI.toast('Tanda terima berhasil disimpan ke arsip.', 'success');

        if (andPrint) {
          A4Renderer.printRecord(res.data);
        }
      },

      refreshKopPreview() {

      updateKopSetting(key, val, rerender = false) {
        const cfg = DataService.getConfig();
        cfg.kop[key] = val;
        DataService.saveConfig(cfg);
        if (rerender) {
          this.renderKopA4View(document.getElementById('main-view'));
          lucide.createIcons({ nodes: [document.getElementById('main-view')] });
        } else {
          this.refreshKopPreview();
        }
      },

      async handleKopImageUpload(input) {
        const file = input.files && input.files[0];
        if (!file) return;
        const check = Security.validateImageFile(file);
        if (!check.ok) {
          UI.toast(check.error, 'error');
          input.value = '';
          return;
        }
        try {
          const { dataUrl, mimeType } = await Security.compressImageToDataUrl(file);
          const cfg = DataService.getConfig();
          cfg.kop.mode = 'image';
          cfg.kop.imageSrc = dataUrl;

          // If GAS mode is enabled, upload to Google Drive and store direct Drive URL
          if (cfg.gas.enabled && cfg.gas.endpointUrl && Auth.getSessionToken()) {
            UI.toast('Mengunggah gambar Kop ke Google Drive...', 'info');
            const driveRes = await DataService.uploadKopToDrive(dataUrl, mimeType, file.name);
            if (driveRes.ok && driveRes.data) {
              cfg.kop.driveFileId = driveRes.data.fileId;
              cfg.kop.imageSrc = driveRes.data.directUrl;
              cfg.kop.imageUrlInput = driveRes.data.directUrl;
              UI.toast('Gambar Kop berhasil disimpan di Google Drive!', 'success');
            } else if (!driveRes.skipped) {
              UI.toast(`Upload Drive gagal (${driveRes.error}). Disimpan di browser lokal.`, 'warning');
            }
          } else {
            UI.toast('Gambar Kop berhasil diterapkan.', 'success');
          }

          const saveRes = DataService.saveConfig(cfg);
          if (!saveRes.ok) UI.toast(saveRes.error, 'error');
          this.renderKopA4View(document.getElementById('main-view'));
          lucide.createIcons({ nodes: [document.getElementById('main-view')] });
        } catch (err) {
          UI.toast(err.message || 'Gagal memproses gambar Kop.', 'error');
        } finally {
          input.value = '';
        }
      },

      applyKopImageUrl() {
        const rawUrl = (document.getElementById('kop-url-input')?.value || '').trim();
        const safe = Security.sanitizeImageSrc(rawUrl);
        if (!safe) {
          UI.toast('URL gambar tidak valid. Gunakan tautan https://...', 'warning');
          return;
        }

        cfg.kop.mode = 'image';
        cfg.kop.imageSrc = safe;
        cfg.kop.imageUrlInput = safe;
